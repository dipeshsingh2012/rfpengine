import requests
from typing import Dict, Any, Optional
from datetime import datetime, timedelta, timezone
from jose import JWTError, jwt
from passlib.context import CryptContext
from google.oauth2 import id_token
from google.auth.transport import requests as google_requests
from sqlalchemy.orm import Session
from fastapi import HTTPException, status
from app.models.user import User

# Configuration (In production, these come from environment variables)
SECRET_KEY = "super-secret-key-change-me"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

class AuthService:
    def __init__(self, db: Session, google_client_id: str, google_client_secret: str):
        self.db = db
        self.google_client_id = google_client_id
        self.google_client_secret = google_client_secret
        self.token_url = "https://oauth2.googleapis.com/token"

    @staticmethod
    def verify_password(plain_password: str, hashed_password: str) -> bool:
        return pwd_context.verify(plain_password, hashed_password)

    @staticmethod
    def get_password_hash(password: str) -> str:
        return pwd_context.hash(password)

    def create_access_token(self, data: dict, expires_delta: Optional[timedelta] = None) -> str:
        to_encode = data.copy()
        if expires_delta:
            expire = datetime.now(timezone.utc) + expires_delta
        else:
            expire = datetime.now(timezone.utc) + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
        to_encode.update({"exp": expire})
        return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    @staticmethod
    def decode_token(token: str) -> Optional[Dict[str, Any]]:
        """Decodes and validates the JWT token."""
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            return payload
        except JWTError:
            return None

    async def exchange_code_for_tokens(self, code: str, redirect_uri: str) -> Dict[str, Any]:
        """Exchanges the authorization code for access and ID tokens."""
        payload = {
            "code": code,
            "client_id": self.google_client_id,
            "client_secret": self.google_client_secret,
            "redirect_uri": redirect_uri,
            "grant_type": "authorization_code",
        }
        
        response = requests.post(self.token_url, data=payload)
        if response.status_code != 200:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Failed to exchange Google code: {response.text}"
            )
        return response.json()

    async def validate_id_token(self, id_token_str: str) -> Dict[str, Any]:
        """Validates the Google ID token."""
        try:
            idinfo = id_token.verify_oauth2_token(
                id_token_str, 
                google_requests.Request(), 
                self.google_client_id
            )
            return idinfo
        except ValueError as e:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail=f"Invalid Google ID token: {str(e)}"
            )

    async def upsert_google_user(self, google_info: Dict[str, Any], tenant_id: str) -> User:
        """
        Upserts the user based on google_id or email within the specific tenant.
        Ensures strict tenant isolation.
        """
        email = google_info.get("email")
        google_id = google_info.get("sub")

        if not email or not google_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Google identity information incomplete."
            )

        # 1. Try to find user by google_id (Global unique identifier)
        user = self.db.query(User).filter(User.google_id == google_id).first()

        if user:
            # SECURITY CHECK: If user exists via google_id, ensure they belong to the requested tenant
            if user.tenant_id != tenant_id:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="User identity mismatch with requested tenant context."
                )
            
            # Update existing user to link google_id if it wasn't linked
            if not user.google_id:
                user.google_id = google_id
                user.auth_provider = "google"
                self.db.commit()
                self.db.refresh(user)
            return user

        # 2. If not found by google_id, try to find by email within the same tenant
        user = self.db.query(User).filter(
            User.email == email, 
            User.tenant_id == tenant_id
        ).first()

        if user:
            # Link the existing local user to Google
            user.google_id = google_id
            user.auth_provider = "google"
            self.db.commit()
            self.db.refresh(user)
            return user
        else:
            # 3. Create new user in this tenant
            new_user = User(
                id=f"user_{google_id}",
                email=email,
                google_id=google_id,
                auth_provider="google",
                tenant_id=tenant_id,
                is_active=True
            )
            self.db.add(new_user)
            self.db.commit()
            self.db.refresh(new_user)
            return new_user

@router.post("/google/callback")
async def google_callback(
    payload: GoogleCallbackRequest,
    x_tenant_id: str = Header(..., alias="X-Tenant-ID"),
    db: Session = Depends(get_db)
):
    """
    Handles the OAuth2 callback. 
    Exchanges code for tokens, validates identity, and returns application JWT.
    """
    auth_service = AuthService(
        db=db, 
        google_client_id=settings.GOOGLE_CLIENT_ID, 
        google_client_secret=settings.GOOGLE_CLIENT_SECRET
    )
    
    # 1. Exchange code for tokens
    tokens = await auth_service.exchange_code_for_tokens(
        code=payload.code, 
        redirect_uri=settings.GOOGLE_REDIRECT_URI
    )
    
    id_token_str = tokens.get("id_token")
    if not id_token_str:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="No ID token returned from Google."
        )

    # 2. Validate ID Token
    google_info = await auth_service.validate_id_token(id_token_str)

    # 3. Upsert User and link to Tenant
    user = await auth_service.upsert_google_user(google_info, x_tenant_id)

    # 4. Return Application JWT (Including tenant_id in payload for verification)
    access_token = auth_service.create_access_token(
        data={"sub": user.id, "email": user.email, "tenant_id": user.tenant_id}
    )

    return {
        "access_token": access_token, 
        "token_type": "bearer",
        "expires_in": 3600,
        "user": {
            "email": user.email,
            "tenant_id": user.tenant_id
        }
    }

@router.get("/verify")
async def verify_auth(
    authorization: Optional[str] = Header(None),
    x_tenant_id: str = Header(..., alias="X-Tenant-ID")
) -> Dict[str, Any]:
    """
    Verifies the JWT token and ensures the user belongs to the requested tenant.
    CRITICAL: Implements strict multi-tenant isolation check.
    """
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Missing or invalid authorization header"
        )
    
    token = authorization.replace("Bearer ", "")
    payload = AuthService.decode_token(token)
    
    if not payload:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Invalid token"
        )

    # CRITICAL FIX: Multi-tenant isolation check
    # Ensure the tenant_id in the token matches the X-Tenant-ID header
    if payload.get("tenant_id") != x_tenant_id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User does not belong to the requested tenant context."
        )

    return {"status": "success", "user": payload}
