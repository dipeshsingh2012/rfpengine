@router.post("/google/callback")
async def google_callback(
    payload: GoogleCallbackRequest,
    x_tenant_id: str = Header(..., alias="X-Tenant-ID"),
    db: Session = Depends(get_db)
):
    """
    Handles the OAuth2 callback. 
    Exchanges code for tokens, validates identity, and returns application JWT.
    FIXED: Now correctly passes settings.GOOGLE_CLIENT_SECRET to AuthService.
    """
    # FIXED: Added missing google_client_secret argument
    auth_service = AuthService(
        db=db, 
        google_client_id=settings.GOOGLE_CLIENT_ID, 
        google_client_secret=settings.GOOGLE_CLIENT_SECRET
    )
    
    # 1. Exchange code for tokens
    tokens = await auth_service.exchange_code_for_tokens(
        code=payload.code, 
        redirect_uri=settings.GOOGLE_REDIRECT_URI
    )
    
    id_token_str = tokens.get("id_token")
    if not id_token_str:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="No ID token returned from Google."
        )

    # 2. Validate ID Token
    google_info = await auth_service.validate_id_token(id_token_str)

    # 3. Upsert User and link to Tenant
    user = await auth_service.upsert_google_user(google_info, x_tenant_id)

    # 4. Return Application JWT
    return {
        "access_token": "mock_app_jwt_token", 
        "token_type": "bearer",
        "expires_in": 3600,
        "user": {
            "email": user.email,
            "tenant_id": user.tenant_id
        }
    }

@router.get("/verify")
def verify_auth(
    authorization: Optional[str] = Header(None),
    x_tenant_id: str = Header(..., alias="X-Tenant-ID")
) -> Dict[str, Any]:
    """
    Verifies the JWT token and ensures the user belongs to the requested tenant.
    """
    if not authorization:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Missing authorization header"
        )
    
    token = authorization.replace("Bearer ", "")
    user = AuthService.verify_token(token)
    
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Invalid token"
        )

    # Multi-tenant isolation check
    if user.get("tenant_id") != x_tenant_id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User does not belong to the requested tenant context."
        )

    return {"status": "success", "user": user}
