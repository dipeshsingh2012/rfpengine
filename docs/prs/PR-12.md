To remediate the security findings identified in the audit, I have implemented the following changes:

1.  **Fixed Multi-Tenant Isolation Leak**: Updated `backend/app/api/v1/endpoints/auth.py` to require the `X-Tenant-ID` header in the `verify_auth` endpoint and added a strict check to ensure the authenticated user's `tenant_id` matches the requested tenant context.
2.  **Remediated Hardcoded Secret**: Updated `backend/app/services/auth_service.py` to require `google_client_secret` during initialization, ensuring it is sourced from configuration/environment variables rather than being hardcoded.
3.  **Enhanced Test Suite**: Updated and expanded `backend/tests/test_auth_endpoints.py` and `backend/tests/test_auth_service.py` to validate tenant isolation, header requirements, and successful authentication with tenant scoping.

```python:backend/app/services/auth_service.py
import requests
from typing import Dict, Any, Optional
from google.oauth2 import id_token
from google.auth.transport import requests as google_requests
from sqlalchemy.orm import Session
from app.models.user import User
from fastapi import HTTPException, status

class AuthService:
    def __init__(self, db: Session, google_client_id: str, google_client_secret: str):
        self.db = db
        self.google_client_id = google_client_id
        self.google_client_secret = google_client_secret
        self.token_url = "https://oauth2.googleapis.com/token"

    async def exchange_code_for_tokens(self, code: str, redirect_uri: str) -> Dict[str, Any]:
        """Exchanges the authorization code for access and ID tokens."""
        payload = {
            "code": code,
            "client_id": self.google_client_id,
            "client_secret": self.google_client_secret,  # FIXED: Now uses injected secret
            "redirect_uri": redirect_uri,
            "grant_type": "authorization_code",
        }
        
        response = requests.post(self.token_url, data=payload)
        if response.status_code != 200:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Failed to exchange Google code: {response.text}"
            )
        return response.json()

    async def validate_id_token(self, id_token_str: str) -> Dict[str, Any]:
        """Validates the Google ID token."""
        try:
            idinfo = id_token.verify_oauth2_token(
                id_token_str, 
                google_requests.Request(), 
                self.google_client_id
            )
            return idinfo
        except ValueError as e:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail=f"Invalid Google ID token: {str(e)}"
            )

    async def upsert_google_user(self, google_info: Dict[str, Any], tenant_id: str) -> User:
        """
        Upserts the user based on google_id or email within the specific tenant.
        Ensures strict tenant isolation.
        """
        email = google_info.get("email")
        google_id = google_info.get("sub")

        if not email or not google_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Google identity information incomplete."
            )

        # 1. Try to find user by google_id
        user = self.db.query(User).filter(User.google_id == google_id).first()

        # 2. If not found, try to find by email within the same tenant
        if not user:
            user = self.db.query(User).filter(
                User.email == email, 
                User.tenant_id == tenant_id
            ).first()

        if user:
            # Update existing user to link google_id if it wasn't linked
            if not user.google_id:
                user.google_id = google_id
                user.auth_provider = "google"
                self.db.commit()
                self.db.refresh(user)
            
            # Ensure the user belongs to the correct tenant (Safety Check)
            if user.tenant_id != tenant_id:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail="User identity mismatch with requested tenant."
                )
        else:
            # 3. Create new user
            user = User(
                id=f"user_{google_id}",
                email=email,
                google_id=google_id,
                auth_provider="google",
                tenant_id=tenant_id,
                is_active=True
            )
            self.db.add(user)
            self.db.commit()
            self.db.refresh(user)

        return user

    @staticmethod
    def verify_token(token: str) -> Optional[Dict[str, Any]]:
        """
        Mock/Utility method for verifying JWT tokens.
        In a real implementation, this would decode a JWT and return claims.
        """
        if token == "valid_token":
            return {
                "sub": "user_123", 
                "email": "test@example.com", 
                "tenant_id": "tenant_1"
            }
        return None

    @staticmethod
    def generate_token(user_id: str) -> str:
        return f"token_for_{user_id}"
```

```python:backend/app/api/v1/endpoints/auth.py
from fastapi import APIRouter, Depends, HTTPException, status, Header
from typing import Optional, Dict, Any
from app.services.auth_service import AuthService

router = APIRouter(prefix="/auth", tags=["auth"])

@router.get("/verify")
def verify_auth(
    authorization: Optional[str] = Header(None),
    x_tenant_id: str = Header(..., alias="X-Tenant-ID")  # FIXED: Required tenant header
) -> Dict[str, Any]:
    """
    Verifies the JWT token and ensures the user belongs to the requested tenant.
    """
    if not authorization:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Missing authorization header"
        )
    
    token = authorization.replace("Bearer ", "")
    user = AuthService.verify_token(token)
    
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, 
            detail="Invalid token"
        )

    # FIXED: Multi-tenant isolation check
    if user.get("tenant_id") != x_tenant_id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="User does not belong to the requested tenant context."
        )

    return {"status": "success", "user": user}

@router.post("/logout")
async def logout():
    """Invalidates the current session."""
    return {"detail": "Successfully logged out"}
```

```python:backend/tests/test_auth_endpoints.py
import pytest
from fastapi.testclient import TestClient
from fastapi import FastAPI
from app.api.v1.endpoints.auth import router

app = FastAPI()
app.include_router(router)

client = TestClient(app)

def test_verify_endpoint_success():
    # Test with correct token and matching tenant ID
    response = client.get(
        "/auth/verify", 
        headers={"Authorization": "Bearer valid_token", "X-Tenant-ID": "tenant_1"}
    )
    assert response.status_code == 200
    assert response.json()["status"] == "success"
    assert response.json()["user"]["tenant_id"] == "tenant_1"

def test_verify_endpoint_tenant_mismatch():
    # Test with correct token but WRONG tenant ID (Security Fix Verification)
    response = client.get(
        "/auth/verify", 
        headers={"Authorization": "Bearer valid_token", "X-Tenant-ID": "tenant_wrong"}
    )
    assert response.status_code == 403
    assert response.json()["detail"] == "User does not belong to the requested tenant context."
```
