import csv
import io
import re
from typing import Any, Dict, Iterator, List, Optional

def sanitize_csv_cell(value: Any) -> str:
    """
    Strip whitespace and escape formula injection characters to prevent 
    CSV Injection (Formula Injection) attacks.
    Consistently strips whitespace before checking dangerous prefixes.
    """
    if value is None:
        return ""
    
    val_str = str(value)
    cleaned = val_str.strip()
    
    dangerous_chars = ('=', '+', '-', '@', '\t', '\r')
    
    if cleaned.startswith(dangerous_chars):
        return f"'{cleaned}"
    
    return val_str

def sanitize_filename_part(part: str) -> str:
    """
    Strictly sanitize filename parts against path traversal (../) 
    and header splitting/injection attacks.
    """
    return re.sub(r"[^a-zA-Z0-9_-]", "", str(part).strip())

def fetch_data_for_tenant(tenant_id: str, resource_id: str) -> List[Dict[str, Any]]:
    """
    SECURE DATA FETCH: Simulates fetching data from a database 
    where the query is strictly bound by the validated tenant_id.
    """
    if not tenant_id or not resource_id:
        return []

    MOCK_DB = {
        "tenant_1": {
            "res_101": [
                {"id": "1", "name": "Tenant 1 Data A", "notes": "Normal note", "val": "100"}, 
                {"id": "2", "name": "Tenant 1 Data B", "notes": "=SUM(A1:A10)", "val": "200"}
            ]
        },
        "tenant_123": {
            "res_101": [
                {"id": "1", "name": "Alice", "email": "alice@example.com", "notes": "=SUM(1,2)"},
                {"id": "2", "name": "Bob", "email": "bob@example.com", "notes": "  -50"},
            ]
        },
        "tenant_2": {
            "res_202": [
                {"id": "1", "name": "Tenant 2 Secret", "notes": "-100", "val": "999"}
            ]
        }
    }
    
    tenant_data = MOCK_DB.get(tenant_id, {})
    return tenant_data.get(resource_id, [])

def generate_csv_chunks(rows: List[Dict[str, Any]], headers: List[str]) -> Iterator[str]:
    """
    Memory-efficient streaming generator that yields CSV rows incrementally.
    """
    output = io.StringIO()
    writer = csv.writer(output)
    
    # 1. Write Header
    writer.writerow(headers)
    yield output.getvalue()
    output.seek(0)
    output.truncate(0)
    
    # 2. Write Rows
    for row in rows:
        sanitized_row = [sanitize_csv_cell(row.get(h, "")) for h in headers]
        writer.writerow(sanitized_row)
        yield output.getvalue()
        output.seek(0)
        output.truncate(0)

import pytest
from fastapi.testclient import TestClient
from fastapi import FastAPI
from backend.app.services.csv_service import (
    sanitize_csv_cell, 
    sanitize_filename_part, 
    generate_csv_chunks, 
    fetch_data_for_tenant
)
from backend.app.api.v1.endpoints.export import router

app = FastAPI()
app.include_router(router)
client = TestClient(app)

def test_sanitize_csv_cell_formula_injection():
    assert sanitize_csv_cell("=SUM(A1:A2)") == "'=SUM(A1:A2)"
    assert sanitize_csv_cell("+123") == "'+123"
    assert sanitize_csv_cell("  -50") == "'-50"
    assert sanitize_csv_cell("@username") == "'@username"
    assert sanitize_csv_cell("\tcontent") == "'\tcontent"
    assert sanitize_csv_cell("normal_text") == "normal_text"
    assert sanitize_csv_cell(123) == "123"
    assert sanitize_csv_cell(None) == ""

def test_sanitize_filename_part_path_traversal():
    assert sanitize_filename_part("../../etc/passwd") == "etcpasswd"
    assert sanitize_filename_part("user/data/file.txt") == "userdatafiletxt"
    assert sanitize_filename_part("tenant_1\r\nX-Injected: True") == "tenant_1XInjectedTrue"
    assert sanitize_filename_part("valid-name_123") == "valid-name_123"

def test_fetch_data_for_tenant_isolation():
    data = fetch_data_for_tenant("tenant_1", "res_101")
    assert len(data) == 2
    assert data[0]["name"] == "Tenant 1 Data A"

    data_cross = fetch_data_for_tenant("tenant_1", "res_202")
    assert data_cross == []

    data_none = fetch_data_for_tenant("unknown", "res_101")
    assert data_none == []

def test_export_endpoint_tenant_isolation():
    response = client.get("/export/csv?resource_id=res_101", headers={"X-Tenant-ID": "tenant_1"})
    assert response.status_code == 200
    assert response.headers["content-type"] == "text/csv"
    assert "attachment; filename=export_tenant_1.csv" in response.headers["content-disposition"]
    assert response.headers["x-tenant-id"] == "tenant_1"

    response_cross = client.get("/export/csv?resource_id=res_202", headers={"X-Tenant-ID": "tenant_1"})
    assert response_cross.status_code == 404

    response_missing = client.get("/export/csv?resource_id=res_101")
    assert response_missing.status_code == 422

def test_generate_csv_chunks():
    headers = ["id", "name", "amount"]
    data = [
        {"id": "1", "name": "Alice", "amount": "100"},
        {"id": "2", "name": "Bob", "amount": "=SUM(A1:A2)"}, 
        {"id": "3", "name": "Charlie", "amount": "  -50"},      
    ]
    
    chunks = list(generate_csv_chunks(data, headers))
    full_output = "".join(chunks)
    
    assert "id,name,amount" in full_output
    assert "1,Alice,100" in full_output
    assert "'=SUM(A1:A2)" in full_output
    assert "'-50" in full_output
